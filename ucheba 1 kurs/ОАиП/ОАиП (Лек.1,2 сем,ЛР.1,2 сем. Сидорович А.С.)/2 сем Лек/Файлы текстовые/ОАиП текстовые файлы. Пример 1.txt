int main()
{
    
   	FILE *f;
	f = fopen(“data.txt", "r");
	if (f==NULL) 
	{	
		printf("Файл не найден... /n Для окончания работы нажмите любую клавишу...");
		return 0;
	}
	
	int n,m;
	
	fscanf(f,"%d",&n);
	fscanf(f,"%d",&m);
	int **a;

	
	a = (int **)calloc (n, sizeof (int *));
	for(int i=0;i<n;i++)
 		a[i] = (int*)calloc (m, sizeof (int));
	
	for(int i=0;i<n;i++)
  		for(int j=0;j<m;j++)
    			fscanf(f,"%d",&a[i][j]);
	
	printf("Прочитана матрица: \n");
	for( int i=0;i<n;i++) 
	{
  		for(int j=0;j<m;j++)
    			printf("%5d",a[i][j]);
  		printf("\n"); 
	}
	return 0;
}
