int num(Ring* p)
{
	int cnt = 0;
	struct Ring* temp = p;
	if (p == NULL) cnt = 0;
	else
		do
		{

			temp = temp->next;
			cnt++;
		} while (temp != p);
		return cnt;

}

}

void push(struct Ring** p, int x) {
	struct Ring* temp;
	temp = new Ring;
	temp->next = NULL;
	temp->data = x;
	if (*p == NULL)
	{
		temp->next = temp;
		*p = temp;

	}
	else
	{
		temp->next = (*p)->next;
		(*p)->next = temp;
	}
}

Ring* push(struct Ring* p, int x) {
	struct Ring* temp;
	temp = new Ring;
	temp->next = NULL;
	temp->data = x;
	if (p == NULL)
	{
		temp->next = temp;
		p = temp;

	}
	else
	{
		temp->next = p->next;
		p->next = temp;
	}
	return p;
}

void pop(struct Ring** p)
{
	if (*p == NULL) {
		cout << "Empty"; return;
	}
	if (*p == (*p)->next)
	{
		delete* p;
		*p = NULL;
	}
	else
	{
		Ring* temp;
		temp = (*p)->next;
		(*p)->next = (*p)->next->next;
		delete temp;
	}
}

Ring* pop(struct Ring* p)
{
	if (p == NULL) {
		cout << "EMPTY"; return NULL;
	}
	if (p == p->next)
	{
		delete p;
		return NULL;
	}
	else
	{
		Ring* temp;
		temp = p->next;
		p->next = p->next->next;
		delete temp;
		return p;
	}
}
