struct FILO {
	int data;
	struct FILO *next;
};

struct FILO *create(struct FILO *, int);  // присоединение элемента к голове, возврат адреса головы
void print(struct FILO *); // просмотр стека
int sum(struct FILO *head);

int main() {
	int a, S;
	struct FILO *head;    // адрес, указывающий на голову стека
	head = NULL;

	while (scanf("%d", &a))
	{
		head = create(head, a);		
	}
	printf("\n");
	print(head);
	S = sum(head);
	printf("\n %d ",S);
	free(head);
}

struct FILO *create(struct FILO *head, int x) {
	struct FILO *n;    // указатель на новую структуру
	n = (struct FILO *)malloc(sizeof(struct FILO)); // выдел€ем пам€ть
	n->next = NULL;
	n->data = x;
	if (head == NULL)
		head = n;
	else
	{
		n->next = head;
		head = n;
	}
	return head;
}


void print(struct FILO *head){
	struct FILO *p = head;
	while (p != NULL) {   // пока не конец стека    
		printf("%2d ---> ", p->data);
		p = p->next;       // продвижение по списку
	}
	printf("\n");
}


int sum(struct FILO *head)
{
	struct FILO *p = head;
	int s = 0;
	while (p != NULL) {   // пока не конец стека    
		s += p->data;
		p = p->next;       // продвижение по списку
	}
	return s;
}
