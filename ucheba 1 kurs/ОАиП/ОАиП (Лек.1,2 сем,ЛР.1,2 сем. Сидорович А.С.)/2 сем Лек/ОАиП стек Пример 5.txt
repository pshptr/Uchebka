struct FILO *push(struct FILO *head, char x) {
	struct FILO *n;    // ÛÍ‡Á‡ÚÂÎ¸ Ì‡ ÌÓ‚Û˛ ÒÚÛÍÚÛÛ
	n = (struct FILO *)malloc(sizeof(struct FILO)); // ‚˚‰ÂÎˇÂÏ Ô‡ÏˇÚ¸
	n->next = NULL;
	n->data = x;
	if (head == NULL)
		head = n;
	else
	{
		n->next = head;
		head = n;
	}
	return head;
}

struct FILO *pop(struct FILO * head)
{
	if (head != NULL)
	{
		head = head->next;
		return head;
	}
