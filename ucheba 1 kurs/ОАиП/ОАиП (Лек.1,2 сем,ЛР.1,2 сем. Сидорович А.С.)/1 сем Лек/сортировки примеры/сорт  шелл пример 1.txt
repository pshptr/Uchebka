#include <iostream>
#include <iomanip>
#include <locale.h>
#include <time.h>
using namespace std;

//Сортировка массива по возрастанию методом Шелла
// Sortirovka massiva po vozrastaniu metodom Shella
int main()
{
	int mas[10], n = 10;
	int i, j;
	srand(time(NULL));
	for (i = 0; i < n; i++)
	{
		mas[i] = rand() % 21;
		cout << setw(4) << mas[i];
	}
	cout << endl;
	cout << endl;
	
	
	
	int gap,flg,t;
	for (gap = n / 2; gap > 0; gap /= 2)
	do {
		flg = 0;
		for (i = 0, j = gap; j < n; i++, j++)
		if (mas[i] > mas[j]) 
		{
			t = mas[j];
			mas[j] = mas[i];
			mas[i] = t;
			flg = 1; 
		}
	} while (flg); 

	

	
	for (i = 0; i < n; i++)
	{
		
		cout << setw(4) << mas[i];
	}
	cout << endl;
}
