#include <iostream>
#include <iomanip>
#include <locale.h>
#include <time.h>
using namespace std;

//Сортировка четных пузырьком
int main()
{
	int mas[10], n = 10;
	int i, j;
	srand(time(NULL));
	for (i = 0; i < n; i++)
	{
		mas[i] = rand() % 21;
		cout << setw(4) << mas[i];
	}
	cout << endl;
	cout << endl;
	for (i = 0; i< n - 1; i++)
	{
		if (mas[i]%2==0)
		for (j = n - 1; j>i; --j)
		{
			if (mas[j] % 2 == 0)
				if (mas[j] > mas[i])
				{
					int t = mas[j];
					mas[j] = mas[i];
					mas[i] = t;
				}
		}
	}


	for (i = 0; i < n; i++)
	{
		
		cout << setw(4) << mas[i];
	}
	cout << endl;
}
