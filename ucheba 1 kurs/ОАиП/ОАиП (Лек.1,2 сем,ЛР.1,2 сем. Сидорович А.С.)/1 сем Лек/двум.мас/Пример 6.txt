# include <stdio.h>
// max el v katoi stroke
// максимальный элемент в к-ой строке матрицы
int main()
{
	int N, M, matr[100][100], i, j, S = 0, k;
	printf("Input size\n");
	scanf("%d%d", &N, &M);

	for (i = 0; i < N; i++)
		for (j = 0; j < M; j++)
			scanf("%d", &matr[i][j]);

	for (i = 0; i < N; i++)
	{
		for (j = 0; j < M; j++)
			printf("%3d", matr[i][j]);
		printf("\n");
	}

	printf("Input number of line\n");
	scanf("%d", &k);

	int max = matr[k-1][0];	
	
		for (j = 0; j < M; j++)
			if (max < matr[k-1][j])
				max = matr[k-1][i];

	printf("max in %d line = %d",k, max);

		return 0;
}
