# include <stdio.h>
// максимальный элемент в матрице
int main()
{
	int N, M, matr[100][100], i, j, S = 0, k;
	printf("Input size\n");
	scanf("%d%d", &N, &M);

	for (i = 0; i < N; i++)
		for (j = 0; j < M; j++)
			scanf("%d", &matr[i][j]);


	int max = matr[0][0];

	
	for (i = 0; i < N; i++)
		for (j = 0; j < M; j++)
			if (max < matr[i][j])
				max = matr[i][j];

	printf("max= %d", max);

		return 0;
}
