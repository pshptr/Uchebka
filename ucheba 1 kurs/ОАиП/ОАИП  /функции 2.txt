int ** input ( int n, int m)
{
	int **a;
	a= mem_init (n,m);
	for (int i=0; i<n; i++)
		for (int j=0; j<m; j++)
			scanf("%d", (*(a+i)+j));
	
	return a;
}




int ** mem_init(int n, int m)
{	
	int **a;	
	a= (int**)calloc(n,sizeof(int*));
	for (int i=0; i<n; i++)
		a[i]= (int*)calloc(m,sizeof(int));
	
}

double mult_odd (int *m, int n)
{
	double p=1;
	for (int i=0; i<n; i++)
		if (odd(*(m+i)==TRUE)
			p*=*(m+i);

	return p;
}

bool odd(int a)
{
	if (a%2!=0)
		return TRUE;
	else return FALSE;
}
