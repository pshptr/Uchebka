#define _CRT_SECURE_NO_WARNINGS
# include <stdio.h>
// ввод -вывод матрицы
int main()
{
	int N, M, matr[100][100], i, j, S = 0;
	printf("Input size\n");
	scanf("%d%d", &N, &M);

	for (i = 0; i < N; i++)
		for (j = 0; j < M; j++)
			scanf("%d", &matr[i][j]);


	for (i = 0; i < N; i++)
	{
		for (j = 0; j < M; j++)
			printf("%3d", matr[i][j]);
		printf("\n");
	}

	return 0;
}
