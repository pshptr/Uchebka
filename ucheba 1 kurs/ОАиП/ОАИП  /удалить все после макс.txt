
# include <stdio.h>
# include <stdlib.h>
# include <alloc.h>

int main()
{
	int *A, n, max,i;
	printf ("input array size\n");
	scanf ("%d", &n);

	A= (int*)calloc (n,sizeof (int));
	
	for (i=0; i<n; i++)
		scanf ("%d", A+i);
	
	max = *A;
	int ind = 0;
	
	for (i=0; i<n; i++)
		if (max <= *(A+i) )
		{
			max= *(A+i);
			ind = i;
		}
	A= (int*) realloc( A, sizeof (int) * (ind +1)); 
	
	n=ind+1;
	for (i=0; i<n; i++)
		printf ("%d", *(A+i));

	printf ("%d", max);
	free(A);

	return 0;
}
