int main()
{
	int N, M, matr[100][100], i, j, S = 0, k;
	printf("Input size\n");
	scanf("%d%d", &N, &M);

	for (i = 0; i < N; i++)
		for (j = 0; j < M; j++)

		for (i = 0; i < N; i++)
			S += matr[i][k-1];
	

	printf("S= %d", S);

		return 0;
}
